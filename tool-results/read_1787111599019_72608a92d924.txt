   124→/* ------------------------------------------------------------------ */
   125→/* Count-up hook                                                       */
   126→/* ------------------------------------------------------------------ */
   127→
   128→function useCountUp(target: number, active: boolean, duration = 1500) {
   129→  const [val, setVal] = useState(0);
   130→  const prefersReduced = useReducedMotion();
   131→
   132→  useEffect(() => {
   133→    if (!active) return;
   134→    if (prefersReduced) {
   135→      const raf = requestAnimationFrame(() => setVal(target));
   136→      return () => cancelAnimationFrame(raf);
   137→    }
   138→    let raf = 0;
   139→    const start = performance.now();
   140→    const tick = (now: number) => {
   141→      const progress = Math.min((now - start) / duration, 1);
   142→      const eased = 1 - Math.pow(1 - progress, 3);
   143→      setVal(target * eased);
   144→      if (progress < 1) raf = requestAnimationFrame(tick);
   145→    };
   146→    raf = requestAnimationFrame(tick);
   147→    return () => cancelAnimationFrame(raf);
   148→  }, [active, target, duration, prefersReduced]);
   149→
   150→  return val;
   151→}

   291→function NewsletterForm() {
   292→  const prefersReduced = useReducedMotion();
   293→  const [email, setEmail] = useState("");
   294→  const [status, setStatus] = useState<"idle" | "loading" | "done">("idle");
   295→
   296→  const onSubmit = (e: React.FormEvent) => {
   297→    e.preventDefault();
   298→    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   299→      toast.error("Please enter a valid email address.");
   300→      return;
   301→    }
   302→    setStatus("loading");
   303→    window.setTimeout(() => {
   304→      setStatus("done");
   305→      toast.success("You're in! 🎉", {
   306→        description: "You'll receive Khidma insights at " + email,
   307→      });
   308→      setEmail("");
   309→      window.setTimeout(() => setStatus("idle"), 3500);
   310→    }, 700);
   311→  };
   312→
   313→  return (
   314→    <div>
   315→      <div className="flex flex-col sm:flex-row gap-3">
   316→        <form onSubmit={onSubmit} className="flex flex-col sm:flex-row gap-3 flex-1">
   317→          <div className="relative flex-1">
   318→            <Input
   319→              type="email"
   320→              value={email}
   321→              onChange={(e) => setEmail(e.target.value)}
   322→              placeholder="you@email.com"
   323→              aria-label="Email address"
   324→              disabled={status === "loading" || status === "done"}
   325→              className="bg-white/5 border-white/15 text-white placeholder:text-white/40 h-12 pr-10 focus-visible:border-[#748684]"
   326→            />
   327→            <AnimatePresence mode="wait">
   328→              {status === "done" && (
   329→                <motion.span
   330→                  key="ok"
   331→                  initial={prefersReduced ? undefined : { opacity: 0, scale: 0.5 }}
   332→                  animate={prefersReduced ? undefined : { opacity: 1, scale: 1 }}
   333→                  exit={prefersReduced ? undefined : { opacity: 0, scale: 0.5 }}
   334→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-emerald-400"
   335→                  aria-hidden
   336→                >
   337→                  <CheckCircle2 className="size-4" />
   338→                </motion.span>
   339→              )}
   340→            </AnimatePresence>
   341→          </div>
   342→          <Button
   343→            type="submit"
   344→            size="lg"
   345→            disabled={status === "loading" || status === "done"}
   346→            className="bg-[#32504d] hover:bg-[#475959] text-white h-12 px-6 group"
   347→          >
   348→            {status === "loading" ? (
   349→              <>
   350→                <span className="size-4 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   351→                Subscribing…
   352→              </>
   353→            ) : status === "done" ? (
   354→              <>
   355→                <CheckCircle2 className="size-4" />
   356→                Subscribed
   357→              </>
   358→            ) : (
   359→              <>
   360→                Subscribe
   361→                <Send className="size-4 transition-transform group-hover:translate-x-0.5" />
   362→              </>
   363→            )}
   364→          </Button>
   365→        </form>
