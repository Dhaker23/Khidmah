   140→/* ---------- filter helpers ---------- */
   141→
   142→function applyFilters(list: Freelancer[], f: FilterState): Freelancer[] {
   143→  const q = f.search.trim().toLowerCase();
   144→  return list.filter((fl) => {
   145→    // search
   146→    if (q) {
   147→      const hay = `${fl.name} ${fl.title} ${fl.skills.join(" ")} ${fl.bio} ${fl.location.city} ${fl.location.country}`.toLowerCase();
   148→      if (!hay.includes(q)) return false;
   149→    }
   150→    // categories — match if any of the freelancer's skills appear in any selected category's skill list
   151→    if (f.categoryIds.length > 0) {
   152→      const allowedSkills = new Set(
   153→        f.categoryIds.flatMap((id) => categories.find((c) => c.id === id)?.skills ?? [])
   154→      );
   155→      const hasMatch = fl.skills.some((s) => allowedSkills.has(s));
   156→      if (!hasMatch) return false;
   157→    }
   158→    // skills (intersection)
   159→    if (f.skills.length > 0) {
   160→      const has = f.skills.some((s) => fl.skills.includes(s));
   161→      if (!has) return false;
   162→    }
   163→    // rate range
   164→    if (fl.hourlyRate < f.rateRange[0] || fl.hourlyRate > f.rateRange[1]) return false;
   165→    // availability
   166→    if (f.availability !== "any" && fl.availability !== f.availability) return false;
   167→    // verifications
   168→    const v = f.verifications;
   169→    if (v.email && !fl.verified.email) return false;
   170→    if (v.phone && !fl.verified.phone) return false;
   171→    if (v.identity && !fl.verified.identity) return false;
   172→    if (v.portfolio && !fl.verified.portfolio) return false;
   173→    if (v.topRated && !fl.topRated) return false;
   174→    // country
   175→    if (f.country !== "all" && fl.location.country !== f.country) return false;
   176→    return true;
   177→  });
   178→}
   179→
   180→function sortFreelancers(list: Freelancer[], key: SortKey): Freelancer[] {
   181→  const copy = [...list];
   182→  switch (key) {
   183→    case "topRated":
   184→      return copy.sort((a, b) => {
   185→        if (a.topRated !== b.topRated) return a.topRated ? -1 : 1;
   186→        return b.rating - a.rating;
   187→      });
   188→    case "newest":
   189→      return copy.sort((a, b) => Number(b.memberSince) - Number(a.memberSince));
   190→    case "mostReviews":
   191→      return copy.sort((a, b) => b.reviewsCount - a.reviewsCount);
   192→    case "rateLow":
   193→      return copy.sort((a, b) => a.hourlyRate - b.hourlyRate);
   194→    case "rateHigh":
   195→      return copy.sort((a, b) => b.hourlyRate - a.hourlyRate);
   196→    default:
   197→      return copy;
   198→  }
   199→}

   226→function FiltersPanel({ filters, setFilters, onClear }: FiltersPanelProps) {
   227→  const toggleArrayValue = (key: "categoryIds" | "skills", value: string) => {
   228→    setFilters((p) => {
   229→      const arr = p[key];
   230→      const next = arr.includes(value)
   231→        ? arr.filter((v) => v !== value)
   232→        : [...arr, value];
   233→      return { ...p, [key]: next };
   234→    });
   235→  };
   236→
   237→  const toggleVerification = (key: keyof FilterState["verifications"]) => {
   238→    setFilters((p) => ({
   239→      ...p,
   240→      verifications: { ...p.verifications, [key]: !p.verifications[key] },
   241→    }));
   242→  };
   243→
   244→  const activeFilterCount =
   245→    filters.categoryIds.length +
   246→    filters.skills.length +
   247→    (filters.availability !== "any" ? 1 : 0) +
   248→    (filters.country !== "all" ? 1 : 0) +
   249→    (filters.rateRange[0] !== RATE_MIN || filters.rateRange[1] !== RATE_MAX ? 1 : 0) +
   250→    Object.values(filters.verifications).filter(Boolean).length;
   251→
   252→  return (
   253→    <div className="space-y-6">
   254→      {/* top row: title + clear */}
   255→      <div className="flex items-center justify-between">
   256→        <div className="flex items-center gap-2">
   257→          <SlidersHorizontal className="size-4 text-[#32504d]" />
   258→          <span className="font-display text-sm font-semibold">Filters</span>
   259→          {activeFilterCount > 0 && (
   260→            <Badge className="bg-[#32504d] text-white hover:bg-[#32504d] text-[10px] h-5 px-1.5">
   261→              {activeFilterCount}
   262→            </Badge>
   263→          )}
   264→        </div>
   265→        <Button
   266→          variant="ghost"
   267→          size="sm"
   268→          onClick={onClear}
   269→          disabled={activeFilterCount === 0}
   270→          className="h-7 text-xs text-muted-foreground hover:text-[#32504d]"
   271→        >
   272→          <RotateCcw className="size-3" />
   273→          Clear all
   274→        </Button>
   275→      </div>
   276→
   277→      <Separator />
   278→
   279→      {/* categories */}
   280→      <FilterSection title="Categories">
   281→        <div className="space-y-2 max-h-64 overflow-y-auto pr-1">
   282→          {categories.map((cat) => {
   283→            const checked = filters.categoryIds.includes(cat.id);
   284→            return (
   285→              <label
   286→                key={cat.id}
   287→                className="flex items-center gap-2.5 cursor-pointer group"
   288→              >
   289→                <Checkbox
   290→                  checked={checked}
   291→                  onCheckedChange={() => toggleArrayValue("categoryIds", cat.id)}
   292→                  className="data-[state=checked]:bg-[#32504d] data-[state=checked]:border-[#32504d]"
   293→                />
   294→                <span className="text-sm flex-1 group-hover:text-[#2b3d3d]">
   295→                  {cat.name}
   296→                </span>
   297→                <Badge variant="outline" className="text-[10px] h-5 px-1.5">
   298→                  {cat.count}
   299→                </Badge>
   300→              </label>
   301→            );
   302→          })}
   303→        </div>
   304→      </FilterSection>
   305→
   306→      <Separator />
   307→
   308→      {/* skills */}
   309→      <FilterSection title="Skills">
   310→        <div className="flex flex-wrap gap-1.5 max-h-44 overflow-y-auto pr-1">
   311→          {ALL_SKILLS.map((skill) => {
   312→            const active = filters.skills.includes(skill);
   313→            return (
   314→              <button
   315→                key={skill}

   317→                onClick={() => toggleArrayValue("skills", skill)}
   318→                className={cn(
   319→                  "text-[11px] px-2.5 py-1 rounded-full border transition-colors",
   320→                  active
   321→                    ? "bg-[#32504d] border-[#32504d] text-white"

   325→                {skill}

   337→          <Slider
   338→            value={filters.rateRange}
   339→            min={RATE_MIN}

   341→            step={RATE_STEP}
   342→            onValueChange={(v) =>
   343→              setFilters((p) => ({

   385→              {opt.label}
   386→            </label>
   387→          ))}
   388→        </RadioGroup>
   389→      </FilterSection>
   390→
   391→      <Separator />

   396→          {VERIFICATION_OPTIONS.map((opt) => {

   415→          })}
   416→        </div>
   417→      </FilterSection>
   418→
   419→      <Separator />
   420→
   421→      {/* country */}
   422→      <FilterSection title="Country">
   423→        <Select
   424→          value={filters.country}
   425→          onValueChange={(v) =>
   426→            setFilters((p) => ({ ...p, country: v as CountryFilter }))

   439→    </div>
   440→  );
   441→}

   591→              <SlidersHorizontal className="size-4" />
   592→            </Button>
   593→          </SheetTrigger>
   594→          <SheetContent
   595→            side="left"
   596→            className="w-[88%] sm:max-w-md overflow-y-auto p-4"
   597→          >
   598→            <SheetHeader>
   599→              <SheetTitle className="font-display">Filter Freelancers</SheetTitle>
   600→              <SheetDescription>
   601→                Narrow down to the perfect talent for your project.
   602→              </SheetDescription>
   603→            </SheetHeader>
   604→            <div className="mt-2">
   605→              <FiltersPanel
   606→                filters={filters}
   607→                setFilters={setFilters}
   608→                onClear={() => {
   609→                  clearFilters();
   610→                  setSheetOpen(false);
   611→                }}
   612→              />
   613→            </div>
   614→          </SheetContent>
   615→        </Sheet>
   616→      </div>
   617→
   618→      <div className="grid grid-cols-1 lg:grid-cols-[280px_1fr] gap-6 lg:gap-8">
   619→        {/* desktop sidebar */}
   620→        <aside className="hidden lg:block">
   621→          <div className="sticky top-24">
   622→            {/* search */}
   623→            <div className="relative mb-5">
   624→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   625→              <Input
   626→                value={filters.search}
   627→                onChange={(e) => updateSearch(e.target.value)}
   628→                placeholder="Search freelancers…"
   629→                className="pl-9 h-10"
   630→              />
   631→              {filters.search && (
   632→                <button
   633→                  onClick={() => updateSearch("")}
   634→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   635→                  aria-label="Clear search"
   636→                >
   637→                  <X className="size-4" />
   638→                </button>
   639→              )}
   640→            </div>
   641→            <Card className="p-5">
   642→              <FiltersPanel
   643→                filters={filters}
   644→                setFilters={setFilters}
   645→                onClear={clearFilters}
   646→              />
   647→            </Card>
   648→          </div>
   649→        </aside>
   650→
   651→        {/* main column */}
   652→        <div>
   653→          {/* toolbar: sort + layout toggle */}
   654→          <div className="flex items-center justify-between gap-3 mb-5">
   655→            <div className="flex items-center gap-2 text-xs text-muted-foreground">
   656→              <span>Sort by</span>
   657→              <Select value={sort} onValueChange={(v) => setSort(v as SortKey)}>
   658→                <SelectTrigger className="h-9 w-[180px] text-sm">
   659→                  <SelectValue>{sortLabel}</SelectValue>
   660→                </SelectTrigger>
   661→                <SelectContent>
   662→                  {SORT_OPTIONS.map((opt) => (
   663→                    <SelectItem key={opt.key} value={opt.key}>
   664→                      {opt.label}
   665→                    </SelectItem>
   666→                  ))}
   667→                </SelectContent>
   668→              </Select>
   669→            </div>
   670→
   671→            <div className="flex items-center gap-1.5">
   672→              {/* layout toggle (dropdown variant for accessibility) */}
   673→              <DropdownMenu>
   674→                <DropdownMenuTrigger asChild>
   675→                  <Button variant="outline" size="sm" className="h-9 px-3">
   676→                    {layout === "grid" ? (
   677→                      <LayoutGrid className="size-4" />
   678→                    ) : (
   679→                      <ListIcon className="size-4" />
   680→                    )}
   681→                    <span className="hidden sm:inline text-xs">
   682→                      {layout === "grid" ? "Grid" : "List"}
   683→                    </span>
   684→                    <ChevronDown className="size-3 opacity-60" />
   685→                  </Button>
   686→                </DropdownMenuTrigger>
   687→                <DropdownMenuContent align="end" className="w-40">
   688→                  <DropdownMenuItem
   689→                    onClick={() => setLayout("grid")}
   690→                    className={cn(
   691→                      "gap-2 cursor-pointer",
   692→                      layout === "grid" && "bg-accent text-[#32504d]"
   693→                    )}
   694→                  >
   695→                    <LayoutGrid className="size-4" />
   696→                    Grid view
   697→                    {layout === "grid" && (
   698→                      <CheckCircle2 className="size-3.5 ml-auto text-[#32504d]" />
   699→                    )}
   700→                  </DropdownMenuItem>
   701→                  <DropdownMenuItem
   702→                    onClick={() => setLayout("list")}
   703→                    className={cn(
   704→                      "gap-2 cursor-pointer",
   705→                      layout === "list" && "bg-accent text-[#32504d]"
   706→                    )}
   707→                  >
   708→                    <ListIcon className="size-4" />
   709→                    List view
   710→                    {layout === "list" && (
   711→                      <CheckCircle2 className="size-3.5 ml-auto text-[#32504d]" />
   712→                    )}
   713→                  </DropdownMenuItem>
   714→                </DropdownMenuContent>
   715→              </DropdownMenu>
   716→            </div>
   717→          </div>
   718→
   719→          {/* results */}
   720→          {visible.length === 0 ? (
   721→            <EmptyState onClear={clearFilters} />
   722→          ) : layout === "grid" ? (
   723→            <motion.div
   724→              initial="hidden"
   725→              animate="show"
   726→              variants={{
   727→                hidden: {},
   728→                show: { transition: { staggerChildren: 0.05 } },
   729→              }}
   730→              className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 gap-5"
   731→            >
   732→              {visible.map((f, i) => (
   733→                <motion.div
   734→                  key={f.id}
   735→                  variants={{
   736→                    hidden: { opacity: 0, y: 14 },
   737→                    show: { opacity: 1, y: 0, transition: { duration: 0.35 } },
   738→                  }}
   739→                >
   740→                  <FreelancerCard freelancer={f} index={0} />
   741→                  <span className="sr-only">{i + 1}</span>
   742→                </motion.div>
   743→              ))}
   744→            </motion.div>
   745→          ) : (
   746→            <motion.div
   747→              initial="hidden"
   748→              animate="show"
   749→              variants={{
   750→                hidden: {},
   751→                show: { transition: { staggerChildren: 0.04 } },
   752→              }}
   753→              className="space-y-3"
   754→            >
   755→              {visible.map((f, i) => (
   756→                <motion.div
   757→                  key={f.id}
   758→                  variants={{
   759→                    hidden: { opacity: 0, y: 8 },
   760→                    show: { opacity: 1, y: 0, transition: { duration: 0.3 } },
   761→                  }}
   762→                >
   763→                  <FreelancerListRow freelancer={f} index={0} />
   764→                  <span className="sr-only">{i + 1}</span>
   765→                </motion.div>
   766→              ))}
   767→            </motion.div>
   768→          )}
   769→
   770→          {/* skeletons + load more (visual only) */}
   771→          {visible.length > 0 && visible.length < sorted.length && (
   772→            <div className="mt-8 flex flex-col items-center gap-6">
   773→              <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 gap-5 w-full">
   774→                {Array.from({ length: 3 }).map((_, i) => (
   775→                  <FreelancerSkeleton key={`sk-${i}`} />
   776→                ))}
   777→              </div>
   778→              <Button
   779→                variant="outline"
   780→                onClick={() => setVisibleCount((c) => c + 6)}
   781→                className="group"
   782→              >
   783→                Load more freelancers
   784→                <ChevronDown className="size-4 transition-transform group-hover:translate-y-0.5" />
   785→              </Button>
   786→            </div>
   787→          )}
   788→
   789→          {visible.length > 0 && visible.length >= sorted.length && (
   790→            <div className="mt-8 flex items-center justify-center gap-2 text-xs text-muted-foreground">
   791→              <Globe className="size-3.5" />
   792→              You&apos;ve reached the end —{" "}
   793→              <span className="font-medium text-foreground">
   794→                {sorted.length}
   795→              </span>{" "}
   796→              verified freelancers
   797→            </div>
   798→          )}
   799→        </div>
   800→      </div>
   801→    </div>
   802→  );
   803→}
