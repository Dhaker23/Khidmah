   125→/* ------------------------------------------------------------------ */
   126→/* Count-up hook                                                       */
   127→/* ------------------------------------------------------------------ */
   128→
   129→function useCountUp(target: number, active: boolean, duration = 1500) {
   130→  const [val, setVal] = useState(0);
   131→  const prefersReduced = useReducedMotion();
   132→
   133→  useEffect(() => {
   134→    if (!active) return;
   135→    if (prefersReduced) {
   136→      const raf = requestAnimationFrame(() => setVal(target));
   137→      return () => cancelAnimationFrame(raf);
   138→    }
   139→    let raf = 0;
   140→    const start = performance.now();
   141→    const tick = (now: number) => {
   142→      const progress = Math.min((now - start) / duration, 1);
   143→      const eased = 1 - Math.pow(1 - progress, 3);
   144→      setVal(target * eased);
   145→      if (progress < 1) raf = requestAnimationFrame(tick);
   146→    };
   147→    raf = requestAnimationFrame(tick);
   148→    return () => cancelAnimationFrame(raf);
   149→  }, [active, target, duration, prefersReduced]);
   150→
   151→  return val;
   152→}

   292→function NewsletterForm() {
   293→  const prefersReduced = useReducedMotion();
   294→  const [email, setEmail] = useState("");
   295→  const [status, setStatus] = useState<"idle" | "loading" | "done">("idle");
   296→
   297→  const onSubmit = (e: React.FormEvent) => {
   298→    e.preventDefault();
   299→    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   300→      toast.error("Please enter a valid email address.");
   301→      return;
   302→    }
   303→    setStatus("loading");
   304→    window.setTimeout(() => {
   305→      setStatus("done");
   306→      toast.success("You're in! 🎉", {
   307→        description: "You'll receive Khidma insights at " + email,
   308→      });
   309→      setEmail("");
   310→      window.setTimeout(() => setStatus("idle"), 3500);
   311→    }, 700);
   312→  };
   313→
   314→  return (
   315→    <div>
   316→      <div className="flex flex-col sm:flex-row gap-3">
   317→        <form onSubmit={onSubmit} className="flex flex-col sm:flex-row gap-3 flex-1">
   318→          <div className="relative flex-1">
   319→            <Input
   320→              type="email"
   321→              value={email}
   322→              onChange={(e) => setEmail(e.target.value)}
   323→              placeholder="you@email.com"
   324→              aria-label="Email address"
   325→              disabled={status === "loading" || status === "done"}
   326→              className="bg-white/5 border-white/15 text-white placeholder:text-white/40 h-12 pr-10 focus-visible:border-[#748684]"
   327→            />
   328→            <AnimatePresence mode="wait">
   329→              {status === "done" && (
   330→                <motion.span
   331→                  key="ok"
   332→                  initial={prefersReduced ? undefined : { opacity: 0, scale: 0.5 }}
   333→                  animate={prefersReduced ? undefined : { opacity: 1, scale: 1 }}
   334→                  exit={prefersReduced ? undefined : { opacity: 0, scale: 0.5 }}
   335→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-emerald-400"
   336→                  aria-hidden
   337→                >
   338→                  <CheckCircle2 className="size-4" />
   339→                </motion.span>
   340→              )}
   341→            </AnimatePresence>
   342→          </div>
   343→          <Button
   344→            type="submit"
   345→            size="lg"
   346→            disabled={status === "loading" || status === "done"}
   347→            className="bg-[#32504d] hover:bg-[#475959] text-white h-12 px-6 group"
   348→          >
   349→            {status === "loading" ? (
   350→              <>
   351→                <span className="size-4 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   352→                Subscribing…
   353→              </>
   354→            ) : status === "done" ? (
   355→              <>
   356→                <CheckCircle2 className="size-4" />
   357→                Subscribed
   358→              </>
   359→            ) : (
   360→              <>
   361→                Subscribe
   362→                <Send className="size-4 transition-transform group-hover:translate-x-0.5" />
   363→              </>
   364→            )}
   365→          </Button>
   366→        </form>
   367→      </div>
   368→      <div className="mt-3 flex items-center gap-1.5 text-[11px] text-white/45">
   369→        <Lock className="size-3" />

   371→      </div>
   372→    </div>

   374→}
