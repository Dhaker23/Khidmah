     5→ * ----------------
     6→ * A premium rotating banner that cycles through 5 featured items every 5s:
     7→ *   1. Featured Freelancer  — "Amira Ben Salah — Full-Stack Developer of the Week"
     8→ *   2. Featured Service     — "I will build a professional Next.js landing page"
     9→ *   3. Featured Job         — "Urgent: Build a Next.js SaaS landing page"

    14→ *  - Full-width dark-teal gradient banner with dot-grid pattern.
    15→ *  - Left column: "FEATURED THIS WEEK" eyebrow + rotating title + description + CTA.
    16→ *  - Right column: rotating visual preview (avatar/cover/event card).
    17→ *  - Bottom row: 5 clickable dot indicators + a thin auto-advance progress bar.
    18→ *
    19→ * Behavior:
    20→ *  - Auto-advances every 5s (full cycle resets progress bar each item).
    21→ *  - Pauses on hover.
    22→ *  - Clicking a dot jumps to that item and resets the timer.
    23→ *  - Respects `prefers-reduced-motion` — no auto-advance, no progress bar,
    24→ *    shows the first item only, dots remain clickable.

    26→ * All animations are framer-motion (slide + fade cross-fade via AnimatePresence).

   601→
   602→// =====================================================================
   603→// Main component
   604→// =====================================================================
   605→
   606→export function FeaturedThisWeek() {
   607→  const prefersReduced = useReducedMotion();
   608→  const items = useFeaturedItems();
   609→  const [index, setIndex] = useState(0);
   610→  const [paused, setPaused] = useState(false);
   611→  const [direction, setDirection] = useState<1 | -1>(1);
   612→  // Progress value (0..1) for the auto-advance bar
   613→  const [progress, setProgress] = useState(0);
   614→  // Ref to the rAF start time so we can compute progress without re-rendering
   615→  const rafRef = useRef<number | null>(null);
   616→  const startRef = useRef<number>(0);
   617→
   618→  const total = items.length;
   619→
   620→  const goTo = useCallback(
   621→    (next: number, dir: 1 | -1 = 1) => {
   622→      const normalized = ((next % total) + total) % total;
   623→      setDirection(dir);
   624→      setIndex(normalized);
   625→      setProgress(0);
   626→    },
   627→    [total]
   628→  );
   629→
   630→  const goNext = useCallback(
   631→    () => goTo(index + 1, 1),
   632→    [goTo, index]
   633→  );
   634→  const goPrev = useCallback(
   635→    () => goTo(index - 1, -1),
   636→    [goTo, index]
   637→  );
   638→
   639→  // -------------------------------------------------------------------
   640→  // Auto-advance loop — drives the progress bar via rAF.
   641→  // Skipped entirely under prefers-reduced-motion (item stays at 0).
   642→  // -------------------------------------------------------------------
   643→  useEffect(() => {
   644→    if (prefersReduced) return;
   645→    if (paused) return;
   646→
   647→    startRef.current = performance.now();
   648→
   649→    const tick = (now: number) => {
   650→      const elapsed = now - startRef.current;
   651→      const p = Math.min(elapsed / ITEM_DURATION_MS, 1);
   652→      setProgress(p);
   653→      if (p >= 1) {
   654→        goNext();
   655→        return; // goNext resets progress; the effect will restart on next render
   656→      }
   657→      rafRef.current = requestAnimationFrame(tick);
   658→    };
   659→
   660→    rafRef.current = requestAnimationFrame(tick);
   661→    return () => {
   662→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
   663→    };
   664→  }, [index, paused, prefersReduced, goNext]);
   665→
   666→  // Clean up on unmount
   667→  useEffect(
   668→    () => () => {
   669→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
   670→    },
   671→    []
   672→  );
   673→
   674→  const current = items[index];
   675→  const KindIcon = KIND_META[current.kind].icon;
   676→
   677→  // Variants — slide + fade. Under reduced-motion we just cross-fade.
   678→  const leftVariants = prefersReduced
   679→    ? {
   680→        initial: { opacity: 0 },
   681→        animate: { opacity: 1 },
   682→        exit: { opacity: 0 },
   683→      }
   684→    : {
   685→        initial: (dir: number) => ({
   686→          opacity: 0,
   687→          x: dir > 0 ? 24 : -24,
   688→        }),
   689→        animate: { opacity: 1, x: 0 },
   690→        exit: (dir: number) => ({
   691→          opacity: 0,
   692→          x: dir > 0 ? -24 : 24,
   693→        }),
   694→      };
   695→
   696→  const rightVariants = prefersReduced
   697→    ? {
   698→        initial: { opacity: 0 },
   699→        animate: { opacity: 1 },
   700→        exit: { opacity: 0 },
   701→      }
   702→    : {
   703→        initial: (dir: number) => ({
   704→          opacity: 0,
   705→          x: dir > 0 ? 40 : -40,
   706→          scale: 0.96,
   707→        }),
   708→        animate: { opacity: 1, x: 0, scale: 1 },
   709→        exit: (dir: number) => ({
   710→          opacity: 0,
   711→          x: dir > 0 ? -40 : 40,
   712→          scale: 0.96,
   713→        }),
   714→      };
   715→
   716→  return (
   717→    <section
   718→      aria-roledescription="carousel"
   719→      aria-label="Featured this week"
   720→      onMouseEnter={() => setPaused(true)}
   721→      onMouseLeave={() => setPaused(false)}
   722→      onFocus={() => setPaused(true)}
   723→      onBlur={() => setPaused(false)}
   724→      className="relative overflow-hidden bg-khidma-gradient"
   725→    >
   726→      {/* Dot grid overlay */}
   727→      <div
   728→        aria-hidden
   729→        className="absolute inset-0 bg-dot-grid opacity-50 pointer-events-none"
   730→      />
   731→      {/* Soft radial glow */}
   732→      <div
   733→        aria-hidden
   734→        className="absolute inset-0 pointer-events-none"
   735→        style={{
   736→          background:
   737→            "radial-gradient(circle at 20% 20%, rgba(116,134,132,0.18) 0%, transparent 50%), radial-gradient(circle at 80% 80%, rgba(50,80,77,0.25) 0%, transparent 55%)",
   738→        }}
   739→      />
   740→
   741→      <div className="relative mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-12 sm:py-16">
   742→        {/* Top row: eyebrow + kind badge */}
   743→        <div className="mb-6 flex items-center justify-between">
   744→          <span className="inline-flex items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.25em] text-white/70">
   745→            <span className="size-1.5 rounded-full bg-emerald-400" />
   746→            Featured this week
   747→          </span>
   748→          <div className="hidden sm:flex items-center gap-1.5">
   749→            <span
   750→              className={cn(
   751→                "inline-flex size-6 items-center justify-center rounded-full",
   752→                KIND_META[current.kind].bg,
   753→                "ring-1",
   754→                KIND_META[current.kind].ring
   755→              )}
   756→            >
   757→              <KindIcon
   758→                className={cn("size-3", KIND_META[current.kind].color)}
   759→              />
   760→            </span>
   761→            <span className="text-[11px] font-medium text-white/70">
   762→              {current.eyebrow}
   763→            </span>
   764→          </div>
   765→        </div>

   827→                    {String(index + 1).padStart(2, "0")} / {String(total).padStart(2, "0")}
   828→                  </span>
   829→                </div>
   830→              </motion.div>
   831→            </AnimatePresence>
   832→          </div>
   833→
   834→          {/* Right: rotating visual */}
   835→          <div className="lg:col-span-5 relative min-h-[280px] sm:min-h-[320px] flex items-center justify-center">
   836→            <AnimatePresence custom={direction} mode="wait">
   837→              <motion.div
   838→                key={current.id}

   840→                initial="initial"

   842→                exit="exit"
   843→                variants={rightVariants}
   844→                transition={SLIDE_TRANSITION}

   847→                {current.visual}
   848→              </motion.div>

   851→        </div>
   852→
   853→        {/* Bottom: dot indicators */}
   854→        <div className="mt-8 sm:mt-10 flex flex-col gap-3">

   856→            {items.map((item, i) => {
   857→              const Icon = KIND_META[item.kind].icon;
   858→              const active = i === index;
   859→              return (
   860→                <button
   861→                  key={item.id}
   862→                  type="button"
   863→                  onClick={() => goTo(i, i > index ? 1 : -1)}
   864→                  aria-label={`Go to ${item.eyebrow}: ${typeof item.title === "string" ? item.title : item.eyebrow}`}
   865→                  aria-current={active ? "true" : undefined}
   866→                  className={cn(
   867→                    "group relative inline-flex h-8 items-center justify-center gap-1.5 rounded-full px-3 transition-all duration-300",
   868→                    active
   869→                      ? "bg-white/15 backdrop-blur-sm ring-1 ring-white/20"
   870→                      : "hover:bg-white/5"
   871→                  )}
   872→                >
   873→                  <Icon
   874→                    className={cn(
   875→                      "size-3 transition-colors",
   876→                      active
   877→                        ? KIND_META[item.kind].color
   878→                        : "text-white/40 group-hover:text-white/70"
   879→                    )}
   880→                  />
   881→                  <span
   882→                    className={cn(
   883→                      "h-1.5 rounded-full transition-all duration-300",
   884→                      active ? "w-6 bg-white" : "w-1.5 bg-white/30"
   885→                    )}
   886→                  />
   887→                </button>
   888→              );
   889→            })}
   890→          </div>
   891→
   892→          {/* Auto-advance progress bar */}
   893→          <div className="relative h-0.5 w-full overflow-hidden rounded-full bg-white/10">
   894→            <motion.div
   895→              className="absolute inset-y-0 left-0 bg-gradient-to-r from-[#748684] via-[#94a8a4] to-white"
   896→              animate={
   897→                prefersReduced
   898→                  ? { width: "0%" }
   899→                  : { width: `${progress * 100}%` }
   900→              }
   901→              transition={prefersReduced ? { duration: 0 } : { duration: 0 }}
   902→              style={{ width: prefersReduced ? "0%" : `${progress * 100}%` }}
   903→              aria-hidden
   904→            />
   905→          </div>
   906→        </div>
   907→      </div>
   908→
   909→      {/* Bottom fade */}
   910→      <div className="absolute inset-x-0 bottom-0 h-px bg-gradient-to-r from-transparent via-white/15 to-transparent" />
   911→    </section>
   912→  );
   913→}
